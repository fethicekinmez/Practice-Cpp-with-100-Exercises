Exercise for reference: The program that takes a string as input and counts the number of vowels and consonants in the string using separate functions.


Answer:


#include <iostream>
#include <string>

using namespace std;

bool isVowel(char c) {
    c = tolower(c);

    if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
        return true;
    } else {
        return false;
    }
}

int countVowels(string str) {
    int count = 0;

    for (int i = 0; i < str.length(); i++) {
        if (isVowel(str[i])) {
            count++;
        }
    }

    return count;
}

int countConsonants(string str) {
    int count = 0;

    for (int i = 0; i < str.length(); i++) {
        if (isalpha(str[i]) && !isVowel(str[i])) {
            count++;
        }
    }

    return count;
}

int main() {
    string str;

    cout << "Enter a string: ";
    getline(cin, str);

    int numVowels = countVowels(str);
    int numConsonants = countConsonants(str);

    cout << "Number of vowels: " << numVowels << endl;
    cout << "Number of consonants: " << numConsonants << endl;

    return 0;
}

The isalpha() function is used in the countConsonants() function to check if a character is an alphabet or not. This function is included in the ctype.h library, which is automatically included when you include iostream.
