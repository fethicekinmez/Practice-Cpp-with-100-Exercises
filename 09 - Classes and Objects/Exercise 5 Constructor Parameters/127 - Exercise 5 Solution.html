Exercise for reference: Program to Illustrate Constructor Parameter

Answer:


#include <iostream>

using namespace std;

class Rectangle {
    private:
        int length, width;
    public:
        Rectangle(int l, int w) {
            length = l;
            width = w;
        }
        int getLength() {
            return length;
        }
        int getWidth() {
            return width;
        }
        int getArea() {
            return length * width;
        }
};

int main() {
    Rectangle r(5, 10);
    cout << "Area of rectangle: " << r.getArea() << endl;
    return 0;
}


Explanation:

The class "Rectangle" has two private member variables, length and width, that are initialized in the constructor. The public member functions provide access to these variables and allow the calculation of the area of the rectangle.
