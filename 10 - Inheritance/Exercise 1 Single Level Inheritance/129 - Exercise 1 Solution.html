Exercise for reference: C++ program that demonstrates single level inheritance using a class hierarchy of vehicles

Answer:


#include <iostream>
using namespace std;

class Vehicle {
  protected:
    string make;
    string model;
    int year;
  public:
    Vehicle(string mk, string mdl, int yr) {
      make = mk;
      model = mdl;
      year = yr;
    }
    void printDetails() {
      cout << "Vehicle Details:\n";
      cout << "Make: " << make << endl;
      cout << "Model: " << model << endl;
      cout << "Year: " << year << endl;
    }
};

class Car: public Vehicle {
  private:
    int numDoors;
  public:
    Car(string mk, string mdl, int yr, int doors): Vehicle(mk, mdl, yr) {
      numDoors = doors;
    }
    void printDetails() {
      Vehicle::printDetails();
      cout << "\nCar Details:\n";
      cout << "Number of Doors: " << numDoors << endl;
    }
};

int main() {
  Vehicle v("Toyota", "Camry", 2022);
  v.printDetails();

  Car c("Toyota", "Camry", 2022, 4);
  c.printDetails();

  return 0;
}


Explanation:

The program creates a class hierarchy of vehicles using single level inheritance. The base class Vehicle has private member variables make, model, and year, and a public member function printDetails() for printing the vehicle's details. The derived class Car inherits from the Vehicle class and has a private member variable numDoors, and a public member function printDetails() for printing the car's details. The derived class accesses the make, model, and year member variables and printDetails() member function from the base class using inheritance.
