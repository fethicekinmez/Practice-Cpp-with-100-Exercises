Exercise for reference: program that demonstrates multiple inheritance using a class hierarchy of a vehicle.


Answer:


#include <iostream>
using namespace std;

class Engine {
  protected:
    string fuelType;
  public:
    Engine(string ft) {
      fuelType = ft;
    }
    void printDetails() {
      cout << "Engine Details:\n";
      cout << "Fuel Type: " << fuelType << endl;
    }
};

class Wheels {
  protected:
    int numWheels;
  public:
    Wheels(int nw) {
      numWheels = nw;
    }
    void printDetails() {
      cout << "\nWheels Details:\n";
      cout << "Number of Wheels: " << numWheels << endl;
    }
};

class Car: public Engine, public Wheels {
  private:
    string brand;
  public:
    Car(string brd, string ft, int nw): Engine(ft), Wheels(nw) {
      brand = brd;
    }
    void printDetails() {
      cout << "\nCar Details:\n";
      cout << "Brand: " << brand << endl;
      Engine::printDetails();
      Wheels::printDetails();
    }
};

int main() {
  Car myCar("Ford", "Gasoline", 4);
  myCar.printDetails();
  return 0;
}


Explanation:

The program creates a class hierarchy of a vehicle using multiple inheritance. The base class Engine has a private member variable fuelType, and a public member function printDetails() for printing the engine's details. The base class Wheels has a private member variable numWheels, and a public member function printDetails() for printing the wheel's details. The derived class Car inherits from both the Engine and Wheels classes and has a private member variable brand, and a public member function printDetails() for printing the car's details. The derived class accesses the fuelType and printDetails() member function from the Engine class, and the numWheels and printDetails() member function from the Wheels class, using multiple inheritance.
