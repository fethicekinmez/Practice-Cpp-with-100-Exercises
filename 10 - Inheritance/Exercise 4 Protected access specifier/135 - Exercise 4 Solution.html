Exercise for reference: program that demonstrates the use of the protected access specifier in inheritance.


Answer:


#include <iostream>
using namespace std;

class Person {
  protected:
    string name;
    int id;
};

class Employee : public Person {
  protected:
    float salary = 60000;
  
  public:
    void setEmployee(string n, int i) {
      name = n;
      id = i;
    }
  
    void printEmployee() {
      cout << "The employee's name is: " << name << endl;
      cout << "The employee's ID number is: " << id << endl;
      cout << "The employee's salary is: $" << salary << endl;
    }
};

int main() {
  Employee emp;
  emp.setEmployee("John", 12345);
  emp.printEmployee();

  return 0;
}


Explanation:

The protected access specifier in inheritance allows derived classes to access the protected members of their base classes. This means that the derived class can access the protected members of its own objects as well as the protected members of objects of its base class.
