Exercise for reference: The program should output the area of a circle and a rectangle, respectively, based on the dimensions of the shapes that were created.


Answer:


#include <iostream>
#include <cmath>
using namespace std;

class Shape {
public:
    virtual float area() {
        return 0;
    }
};

class Circle : public Shape {
private:
    float radius;
public:
    Circle(float r) {
        radius = r;
    }
    float area() {
        return M_PI * radius * radius;
    }
};

class Rectangle : public Shape {
private:
    float length, width;
public:
    Rectangle(float l, float w) {
        length = l;
        width = w;
    }
    float area() {
        return length * width;
    }
};

int main() {
    float radius, length, width;

    cout << "Enter radius of the circle: ";
    cin >> radius;

    cout << "Enter length and width of the rectangle: ";
    cin >> length >> width;

    Circle c(radius);
    Rectangle r(length, width);

    cout << "Area of the circle: " << c.area() << endl;
    cout << "Area of the rectangle: " << r.area() << endl;

    return 0;
}


Explanation:

In the example code provided, we define a base class Shape with a virtual function area(), which returns the area of a shape. We then derive two classes Circle and Rectangle from the Shape class and implement the area() function in each derived class.

In the main program, we create objects of each class and call the area() function on each object. Since the area() function is declared as virtual in the Shape class and is overridden in the derived classes, the correct version of the function is called at run-time based on the actual type of object being referred to. This is an example of run-time polymorphism.
