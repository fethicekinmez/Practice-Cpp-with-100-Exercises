Exercise for reference: program to sort the contents of a file alphabetically

Answer:


#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main() {
    ifstream infile("example.txt");

    if (!infile) {
        cerr << "Error: Unable to open file!" << endl;
        return 1;
    }

    string lines[100];
    int count = 0;
    string line;

    while (getline(infile, line)) {
        lines[count++] = line;
    }

    infile.close();

    for (int i = 0; i < count; i++) {
        for (int j = i+1; j < count; j++) {
            if (lines[i] > lines[j]) {
                string temp = lines[i];
                lines[i] = lines[j];
                lines[j] = temp;
            }
        }
    }

    ofstream outfile("example_sorted.txt");

    if (!outfile) {
        cerr << "Error: Unable to create file!" << endl;
        return 1;
    }

    for (int i = 0; i < count; i++) {
        outfile << lines[i] << endl;
    }

    outfile.close();

    cout << "File sorted successfully." << endl;

    return 0;
}


Explanation:

This program reads the contents of a file named example.txt into an array of strings, sorts the array using nested loops, and writes the sorted lines to a new file named example_sorted.txt. It first opens the input file using the ifstream class and checks if the file was opened successfully. If not, it prints an error message and returns an error code. Otherwise, the program reads the lines of the file into an array of strings using a loop that keeps track of the number of lines read. It then sorts the array of strings using nested loops and closes the input file. Next, it opens the output file using the ofstream class and checks if the file was opened successfully. If not, it prints an error message and returns an error code. Finally, the program writes each line of the sorted array to the output file using another loop and closes the output file. It then prints a message indicating that the file was sorted successfully.
