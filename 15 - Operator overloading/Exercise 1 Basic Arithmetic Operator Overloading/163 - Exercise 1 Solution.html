Exercise for reference: Vector class has overloaded the +, -, and * operators to add, subtract, and multiply vectors, respectively.

Answer:


#include <iostream>
using namespace std;

class Vector {
public:
    int x, y, z;
    
    Vector operator+ (const Vector& other) const {
        Vector result;
        result.x = x + other.x;
        result.y = y + other.y;
        result.z = z + other.z;
        return result;
    }
    
    Vector operator- (const Vector& other) const {
        Vector result;
        result.x = x - other.x;
        result.y = y - other.y;
        result.z = z - other.z;
        return result;
    }
    
    Vector operator* (const int scalar) const {
        Vector result;
        result.x = x * scalar;
        result.y = y * scalar;
        result.z = z * scalar;
        return result;
    }
};

int main() {
    Vector v1 {1, 2, 3};
    Vector v2 {4, 5, 6};
    
    Vector v3 = v1 + v2;
    Vector v4 = v2 - v1;
    Vector v5 = v1 * 3;
    
    cout << "v1 + v2 = (" << v3.x << ", " << v3.y << ", " << v3.z << ")\n";
    cout << "v2 - v1 = (" << v4.x << ", " << v4.y << ", " << v4.z << ")\n";
    cout << "v1 * 3 = (" << v5.x << ", " << v5.y << ", " << v5.z << ")\n";
    
    return 0;
}


Explanation:

This program defines a Vector class with x, y, and z components. It then overloads the +, -, and * operators to allow vectors to be added, subtracted, and multiplied by scalars, respectively.

In the main function, the program creates two Vector objects v1 and v2, and then performs the overloaded operations on them to create new Vector objects v3, v4, and v5. The program then prints out the results of these operations.
