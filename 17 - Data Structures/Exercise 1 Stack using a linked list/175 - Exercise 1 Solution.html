Exercise for reference: The program to implementing a stack using a linked list

Answer:


#include <iostream>
using namespace std;

// Define the node structure for the linked list
struct Node {
    int data;
    Node* next;
};

class Stack {
    private:
        Node* top;
    public:
        Stack() { // Constructor to initialize the stack
            top = NULL;
        }
        void push(int val) { // Function to push an element onto the stack
            Node* newNode = new Node;
            newNode->data = val;
            newNode->next = top;
            top = newNode;
        }
        void pop() { // Function to pop an element from the stack
            if (top == NULL) {
                cout << "Stack Underflow" << endl;
                return;
            }
            Node* temp = top;
            top = top->next;
            delete temp;
        }
        void display() { // Function to display the stack
            if (top == NULL) {
                cout << "Stack is empty" << endl;
                return;
            }
            Node* temp = top;
            while (temp != NULL) {
                cout << temp->data << " ";
                temp = temp->next;
            }
            cout << endl;
        }
};

int main() {
    Stack s; // Create a new stack object
    s.push(5); // Push some elements onto the stack
    s.push(10);
    s.push(15);
    s.display(); // Display the stack
    s.pop(); // Pop an element from the stack
    s.display(); // Display the stack again
    return 0;
}
