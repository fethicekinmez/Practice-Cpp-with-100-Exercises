Exercise for reference: The program should allow the user to insert, delete, and display elements in the doubly linked list.

Answer:


#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* prev;
    Node* next;
};

class DoublyLinkedList {
    private:
        Node* head;
        Node* tail;
    public:
        DoublyLinkedList() {
            head = NULL;
            tail = NULL;
        }
        void insertAtBeginning(int val) {
            Node* newNode = new Node;
            newNode->data = val;
            newNode->prev = NULL;
            if (head == NULL) {
                tail = newNode;
                newNode->next = NULL;
            } else {
                newNode->next = head;
                head->prev = newNode;
            }
            head = newNode;
            cout << "Element inserted successfully." << endl;
        }
        void insertAtEnd(int val) {
            Node* newNode = new Node;
            newNode->data = val;
            newNode->next = NULL;
            if (head == NULL) {
                newNode->prev = NULL;
                head = newNode;
            } else {
                tail->next = newNode;
                newNode->prev = tail;
            }
            tail = newNode;
            cout << "Element inserted successfully." << endl;
        }
        void insertAtPosition(int val, int pos) {
            Node* newNode = new Node;
            newNode->data = val;
            if (head == NULL) {
                newNode->prev = NULL;
                newNode->next = NULL;
                head = newNode;
                tail = newNode;
                cout << "Element inserted successfully." << endl;
                return;
            }
            Node* temp = head;
            for (int i = 1; i < pos && temp != NULL; i++) {
                temp = temp->next;
            }
            if (temp == NULL) {
                cout << "Position out of range." << endl;
                return;
            }
            if (temp == head) {
                newNode->prev = NULL;
                newNode->next = head;
                head->prev = newNode;
                head = newNode;
            } else {
                newNode->next = temp;
                newNode->prev = temp->prev;
                temp->prev->next = newNode;
                temp->prev = newNode;
            }
            cout << "Element inserted successfully." << endl;
        }
        void deleteAtPosition(int pos) {
            if (head == NULL) {
                cout << "List is empty." << endl;
               
            return;
        }
        Node* temp = head;
        for (int i = 1; i < pos && temp != NULL; i++) {
            temp = temp->next;
        }
        if (temp == NULL) {
            cout << "Position out of range." << endl;
            return;
        }
        if (temp == head && temp == tail) {
            head = NULL;
            tail = NULL;
        } else if (temp == head) {
            head = temp->next;
            head->prev = NULL;
        } else if (temp == tail) {
            tail = temp->prev;
            tail->next = NULL;
        } else {
            temp->prev->next = temp->next;
            temp->next->prev = temp->prev;
        }
        delete temp;
        cout << "Element deleted successfully." << endl;
        return;
    }
    void display() {
        if (head == NULL) {
            cout << "List is empty." << endl;
            return;
        }
        cout << "Doubly Linked List: ";
        Node* temp = head;
        while (temp != NULL) {
            cout << temp->data << " ";
            temp = temp->next;
        }
        cout << endl;
    }
};

int main() {
    DoublyLinkedList dll;
    int choice, value, position;
    while (true) {
        cout << "---- Doubly Linked List Implementation ----" << endl;
        cout << "1. Insert at beginning" << endl;
        cout << "2. Insert at end" << endl;
        cout << "3. Insert at position" << endl;
        cout << "4. Delete at position" << endl;
        cout << "5. Display" << endl;
        cout << "6. Exit" << endl;
        cout << "Enter your choice: ";
        cin >> choice;
        switch (choice) {
            case 1:
                cout << "Enter the value to be inserted: ";
                cin >> value;
                dll.insertAtBeginning(value);
                break;
            case 2:
                cout << "Enter the value to be inserted: ";
                cin >> value;
                dll.insertAtEnd(value);
                break;
            case 3:
                cout << "Enter the value to be inserted: ";
                cin >> value;
                cout << "Enter the position to insert: ";
                cin >> position;
                dll.insertAtPosition(value, position);
                break;
            case 4:
                cout << "Enter the position to delete: ";
                cin >> position;
                dll.deleteAtPosition(position);
                break;
            case 5:
                dll.display();
                break;
            case 6:
                cout << "Exiting the program..." << endl;
                return 0;
            default:
                cout << "Invalid choice." << endl;
        }
    }
    return 0;
}


Explanation:

The program starts by defining a Node struct that contains the data value and pointers to the previous and next nodes. A DoublyLinkedList class is then defined that contains a pointer to the head and tail nodes.

The insertAtBeginning function inserts a new node at the beginning of the list. If the list is empty, the new node becomes both the head and tail. Otherwise, the new node becomes the new head and its next pointer is set to the previous head. The previous head's prev pointer is set to the new node.

The insertAtEnd function inserts a new node at the end of the list. If the list is empty, the new node becomes both the head and tail. Otherwise, the new node becomes the new tail and its prev pointer is set to the previous tail. The previous tail's next pointer is set to the new node.

The insertAtPosition function inserts a new node at a specified position in the list. If the specified position is 1, the new node is inserted at the beginning of the list using the insertAtBeginning function. If the specified position is greater than the current size of the list, the new node is inserted at the end of the list using the insertAtEnd function. Otherwise, the list is traversed to find the node at the specified position. The new node's prev pointer is set to the previous node's prev pointer and the new node's next pointer is set to the previous node. The previous node's prev pointer is set to the new node and the previous node's next pointer is set to the new node.

The deleteAtPosition function deletes a node at a specified position in the list. If the specified position is 1, the head is set to the second node and the new head's prev pointer is set to NULL. If the specified position is equal to the current size of the list, the tail is set to the second-to-last node and the new tail's next pointer is set to NULL. Otherwise, the list is traversed to find the node at the specified position. The previous node's next pointer is set to the next node and the next node's prev pointer is set to the previous node.

The display function displays all elements in the list. It traverses the list starting from the head and prints each node's data value.

The main function creates a DoublyLinkedList object and provides a menu for the user to interact with the list. The user can insert nodes at the beginning, end, or a specified position, delete nodes at a specified position, and display the list. The program loops until the user chooses to exit.
