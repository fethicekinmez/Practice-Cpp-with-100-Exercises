Exercise for reference: The client should send a message to the server and receive a response message from the server.

Server side code:


#include <iostream>
#include <string.h>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <unistd.h>

using namespace std;

int main(int argc, char const *argv[]) {
    int socket_fd, client_fd;
    struct sockaddr_in server_addr, client_addr;
    char buffer[1024] = {0};
    const char *response = "Message received!";

    // Create socket file descriptor
    if ((socket_fd = socket(AF_INET, SOCK_STREAM, 0)) == 0) {
        perror("socket failed");
        exit(EXIT_FAILURE);
    }

    // Bind socket to localhost:8080
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = INADDR_ANY;
    server_addr.sin_port = htons(8080);

    if (bind(socket_fd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0) {
        perror("bind failed");
        exit(EXIT_FAILURE);
    }

    // Listen for incoming connections
    if (listen(socket_fd, 3) < 0) {
        perror("listen failed");
        exit(EXIT_FAILURE);
    }

    cout << "Waiting for incoming connections..." << endl;

    // Accept incoming connections
    socklen_t client_addr_len = sizeof(client_addr);
    if ((client_fd = accept(socket_fd, (struct sockaddr *)&client_addr, &client_addr_len)) < 0) {
        perror("accept failed");
        exit(EXIT_FAILURE);
    }

    cout << "Client connected." << endl;

    // Receive message from client
    int message_len;
    if ((message_len = recv(client_fd, buffer, 1024, 0)) == -1) {
        perror("recv failed");
        exit(EXIT_FAILURE);
    }

    buffer[message_len] = '\0';
    cout << "Message received from client: " << buffer << endl;

    // Send response to client
    if (send(client_fd, response, strlen(response), 0) == -1) {
        perror("send failed");
        exit(EXIT_FAILURE);
    }

    close(client_fd);
    close(socket_fd);

    return 0;
}
