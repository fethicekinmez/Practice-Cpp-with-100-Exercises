Exercise for reference: Use a switch statement to check the user's choice and call the appropriate member function of the operation class.

Answer:


#include <iostream>
using namespace std;

class operation
{
    int a, b, add, sub, mul;
    float div;

public:
    void get();
    void sum();
    void difference();
    void product();
    void division();
};

inline void operation ::get()
{
    cout << "Enter first value:";
    cin >> a;
    cout << "Enter second value:";
    cin >> b;
}

inline void operation ::sum()
{
    add = a + b;
    cout << "Addition of two numbers: " << a + b << "\n";
}

inline void operation ::difference()
{
    sub = a - b;
    cout << "Difference of two numbers: " << a - b << "\n";
}

inline void operation ::product()
{
    mul = a * b;
    cout << "Product of two numbers: " << a * b << "\n";
}

inline void operation ::division()
{
    div = a / b;
    cout << "Division of two numbers: " << a / b << "\n";
}

int main()
{
    cout << "Program using switch statement\n";
    operation s;
    s.get();

    int choice;
    cout << "Which operation do you want to perform? \n";
    cout << "1. Addition \n2. Subtraction \n3. Multiplication \n4. Division \n";
    cout << "Enter your choice: ";
    cin >> choice;

    switch (choice)
    {
        case 1:
            s.sum();
            break;
        case 2:
            s.difference();
            break;
        case 3:
            s.product();
            break;
        case 4:
            s.division();
            break;
        default:
            cout << "Invalid choice";
    }

    return 0;
}


Explanation:

In the modified code, we added a switch statement to allow the user to choose which operation they want to perform on the two input values. The get() method is called to get the input values, and then the user is prompted to enter their choice of operation. Depending on their choice, the appropriate member function of the operation class is called using a switch statement. If the user enters an invalid choice, an error message is displayed.
