Exercise for reference: program that demonstrates string manipulation using library functions:


Answer:


#include <iostream>
#include <string>

using namespace std;

int main() {
  string str = "Hello, World!";
  
  // print the original string
  cout << "Original string: " << str << endl;
  
  // get the length of the string
  int length = str.length();
  cout << "Length of the string: " << length << endl;
  
  // convert the string to uppercase
  for (int i = 0; i < length; i++) {
    str[i] = toupper(str[i]);
  }
  cout << "Uppercase string: " << str << endl;
  
  // convert the string to lowercase
  for (int i = 0; i < length; i++) {
    str[i] = tolower(str[i]);
  }
  cout << "Lowercase string: " << str << endl;
  
  // find the position of a character in the string
  char search_char = 'o';
  size_t pos = str.find(search_char);
  if (pos != string::npos) {
    cout << "Position of character '" << search_char << "' in string: " << pos << endl;
  } else {
    cout << "Character not found in string." << endl;
  }
  
  return 0;
}


Explanation:

  • First, we declare a string variable str and initialize it with the string "Hello, World!".

  • Next, we use the cout statement to print the original string to the console.

  • We then use the length function to find the length of the string and print it to the console.

  • Next, we use a for loop to convert the string to uppercase using the toupper function and print it to the console.

  • We then use another for loop to convert the string back to lowercase using the tolower function and print it to the console.

  • Finally, we use the find function to find the position of the character 'o' in the string and print it to the console.
