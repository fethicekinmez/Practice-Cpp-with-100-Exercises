Exercise for reference: The program should display a table of the students' names, ages, and grades entered by the user.


Answer:


#include <iostream>
#include <string>

using namespace std;

struct Student {
    string name;
    int age;
    int grade;
};

int main() {
    int num_students;
    cout << "Enter the number of students: ";
    cin >> num_students;

    Student students[num_students];

    for (int i = 0; i < num_students; i++) {
        cout << "\nEnter the details of student " << i+1 << ":\n";
        cout << "Name: ";
        cin >> students[i].name;
        cout << "Age: ";
        cin >> students[i].age;
        cout << "Grade: ";
        cin >> students[i].grade;
    }

    cout << "\nTable of student details:\n";
    cout << "----------------------------------\n";
    cout << "| Name         | Age   | Grade   |\n";
    cout << "----------------------------------\n";
    for (int i = 0; i < num_students; i++) {
        cout << "| " << students[i].name << "     ";
        cout << "| " << students[i].age << "    ";
        cout << "| " << students[i].grade << "      |\n";
    }
    cout << "----------------------------------\n";

    return 0;
}

We use the std namespace and create a struct called Student to store information about each student:

In the main function, we prompt the user to enter the number of students and create an array of Student structs with the size specified by the user:

Next, we use a for loop to prompt the user to enter the name, age, and grade of each student and store the information in the corresponding fields of each Student struct in the array:

Finally, we use another for loop to display the table of student details:

The output is formatted to display the information in a table with columns for the student's name, age, and grade.
